The following example integrates two AI APIs (OpenAI and Gemini). The same approach can be extended to other AI tools.

## import requests

### API Keys
OPENAI_API_KEY = "your_openai_api_key"
GEMINI_API_KEY = "your_gemini_api_key"

prompt = "Explain the benefits of Artificial Intelligence in education."

```py
# OpenAI API Call
def get_openai_response(prompt):

    url = "https://api.openai.com/v1/chat/completions"

    headers = {
        "Authorization": f"Bearer {OPENAI_API_KEY}",
        "Content-Type": "application/json"
    }

    payload = {
        "model": "gpt-4",
        "messages": [
            {"role": "user", "content": prompt}
        ]
    }

    response = requests.post(url,
                             headers=headers,
                             json=payload)

    return response.json()["choices"][0]["message"]["content"]

# Gemini API Call
def get_gemini_response(prompt):

    url = f"https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent?key={GEMINI_API_KEY}"

    payload = {
        "contents": [
            {
                "parts": [
                    {"text": prompt}
                ]
            }
        ]
    }

    response = requests.post(url, json=payload)

    return response.json()["candidates"][0]["content"]["parts"][0]["text"]

# Fetch Responses
openai_output = get_openai_response(prompt)
gemini_output = get_gemini_response(prompt)

# Display Outputs
print("\nOPENAI RESPONSE:\n")
print(openai_output)

print("\nGEMINI RESPONSE:\n")
print(gemini_output)

# Compare Outputs
print("\nCOMPARISON REPORT")

print("OpenAI Length:", len(openai_output))
print("Gemini Length:", len(gemini_output))

if len(openai_output) > len(gemini_output):
    print("OpenAI generated a more detailed response.")
else:
    print("Gemini generated a more detailed response.")
```
